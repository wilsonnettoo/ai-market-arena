## Global Constraints

Valem integralmente as do plano M0–M1. Reforço dos pontos que este marco exercita:

- **`Adjustment.RAW` explícito em toda requisição de barras.** O parâmetro é opcional na API; deixá-lo implícito é como o erro entra por omissão.
- **Nenhum float em registro hasheado.** Preços e probabilidades viajam como string, formatados com `Decimal.quantize`.
- **`datetime` sempre com timezone.** `ruff` (regra `DTZ`) reprova o contrário.
- **Calendário de pregão via `exchange_calendars`**, nunca contagem de dias corridos.
- **Toda resposta de API é arquivada write-once com `ingestion_time` antes de ser interpretada.** É o único item do projeto cujo custo de retrofit é infinito.
- **Dia sem publicação é `OutageRecord`.** Um buraco no histórico público é indistinguível de uma decisão omitida.

# arena/storage/archive.py
"""Arquivo write-once de toda resposta de API.

Dado nao coletado nao existe. Este e o unico modulo do projeto cujo custo de
retrofit e infinito, e por isso ele grava ANTES de qualquer interpretacao:
se o parser estiver errado, o byte original continua la.
"""

from __future__ import annotations

import gzip
import hashlib
from pathlib import Path

_EXT = {
    "application/json": "json",
    "text/csv": "csv",
    "text/plain": "txt",
}


class ArchiveExists(RuntimeError):
    """Tentativa de sobrescrever arquivo bruto. O arquivo e write-once."""


def _safe_stamp(ingestion_time_utc: str) -> str:
    return ingestion_time_utc.replace(":", "-")


def archive_raw(
    root: Path,
    source: str,
    session_date: str,
    ingestion_time_utc: str,
    payload: bytes,
    content_type: str,
) -> dict:
    ext = _EXT.get(content_type)
    if ext is None:
        raise ValueError(f"content_type nao suportado: {content_type}")

    rel = f"data/raw/{source}/{session_date}/{_safe_stamp(ingestion_time_utc)}.{ext}.gz"
    destino = root / rel
    if destino.exists():
        raise ArchiveExists(rel)

    destino.parent.mkdir(parents=True, exist_ok=True)
    # mtime=0 mantem o gzip byte-identico para o mesmo payload
    with gzip.GzipFile(filename="", mode="wb", fileobj=destino.open("wb"), mtime=0) as fh:
        fh.write(payload)

    return {
        "source": source,
        "session_date": session_date,
        "ingestion_time_utc": ingestion_time_utc,
        "path": rel,
        "sha256": hashlib.sha256(payload).hexdigest(),
        "bytes": len(payload),
        "content_type": content_type,
    }


def read_raw(root: Path, rel_path: str) -> bytes:
    with gzip.open(root / rel_path, "rb") as fh:
        return fh.read()
```

# arena/ingest/calendar_us.py
"""Calendario de pregao americano.

Contar dias corridos e a origem garantida de bug de fronteira de dia,
especialmente para quem opera de Brasilia: o horario de verao americano
muda em datas diferentes do brasileiro.
"""

from __future__ import annotations

from datetime import date
from functools import lru_cache

import exchange_calendars as xcals
import pandas as pd

CALENDAR_NAME = "XNYS"


@lru_cache(maxsize=1)
def _cal():
    return xcals.get_calendar(CALENDAR_NAME, start="2000-01-01", end="2035-12-31")


def is_session(d: date) -> bool:
    return bool(_cal().is_session(pd.Timestamp(d)))


def previous_session(d: date) -> date:
    return _cal().previous_session(pd.Timestamp(d)).date()


def session_offset(d: date, n: int) -> date:
    if not is_session(d):
        raise ValueError(f"{d} nao e sessao de pregao")
    if n < 1:
        raise ValueError("n precisa ser >= 1")
    atual = pd.Timestamp(d)
    for _ in range(n):
        atual = _cal().next_session(atual)
    return atual.date()


def last_n_sessions(end: date, n: int) -> list[date]:
    if n < 1:
        raise ValueError("n precisa ser >= 1")
    fim = pd.Timestamp(end) if is_session(end) else pd.Timestamp(previous_session(end))
    sessoes = [fim]
    for _ in range(n - 1):
        sessoes.append(_cal().previous_session(sessoes[-1]))
    return sorted(s.date() for s in sessoes)
```

# arena/storage/bars.py
"""Store de barras diarias em Parquet.

Preco e gravado BRUTO. A tabela de acoes corporativas entra no M4 junto com o
Ledger; em M2 nada precisa dela, porque a resolucao de previsao anula a
observacao quando detecta variacao diaria acima de 25% na janela (Task 16).
"""

from __future__ import annotations

from datetime import date, datetime
from decimal import Decimal
from pathlib import Path

import pandas as pd

_COLS = ["symbol", "session_date", "open", "high", "low", "close", "volume"]
_PRICE_COLS = ["open", "high", "low", "close", "volume"]


def write_bars(root: Path, rows: list[dict]) -> Path:
    if not rows:
        raise ValueError("nenhuma barra para gravar")

    sessoes = {r["session_date"] for r in rows}
    if len(sessoes) != 1:
        raise ValueError(f"write_bars aceita uma sessao por chamada, recebeu {sorted(sessoes)}")

    for r in rows:
        for c in _PRICE_COLS:
            if not isinstance(r[c], str):
                raise TypeError(f"{c} precisa ser str, recebeu {type(r[c]).__name__}")

    sessao = sessoes.pop()
    destino = root / "data" / "bars" / f"{sessao}.parquet"
    destino.parent.mkdir(parents=True, exist_ok=True)
    pd.DataFrame(rows, columns=_COLS).to_parquet(destino, index=False)
    return destino


def _to_date(v) -> date:
    if isinstance(v, date) and not isinstance(v, datetime):
        return v
    return pd.Timestamp(v).date()


def read_bars_as_of(root: Path, symbols: list[str], sessions: list[date]) -> pd.DataFrame:
    quadros = []
    for s in sorted(sessions):
        arquivo = root / "data" / "bars" / f"{s.isoformat()}.parquet"
        if arquivo.exists():
            quadros.append(pd.read_parquet(arquivo))

    if not quadros:
        return pd.DataFrame(columns=symbols)

    bruto = pd.concat(quadros, ignore_index=True)
    bruto = bruto[bruto["symbol"].isin(symbols)]
    largo = bruto.pivot(index="session_date", columns="symbol", values="close")
    largo.index = [_to_date(i) for i in largo.index]
    largo = largo.reindex(columns=symbols)
    return largo.map(lambda v: Decimal(v) if isinstance(v, str) else None).sort_index()
```

# arena/ingest/alpaca_eod.py
"""Barras diarias com Adjustment.RAW.

O parametro `adjustment` e opcional na API do Alpaca. Deixa-lo implicito e como
o preco ajustado entra no registro por omissao — e o ajustado nao serve para um
registro auditavel, porque ele muda retroativamente a cada acao corporativa.

`asof` faz o mapeamento historico de ticker: sem ele, um papel que trocou de
simbolo aparece sob o nome de hoje em dados de meses atras.
"""

from __future__ import annotations

import json
import os
from datetime import date, datetime, time, timezone
from decimal import Decimal

from alpaca.data.enums import Adjustment, DataFeed
from alpaca.data.historical import StockHistoricalDataClient
from alpaca.data.requests import StockBarsRequest
from alpaca.data.timeframe import TimeFrame

_Q = Decimal("0.0001")


class AlpacaCredentialsMissing(RuntimeError):
    pass


def _client() -> StockHistoricalDataClient:
    key = os.environ.get("ALPACA_API_KEY_ID")
    secret = os.environ.get("ALPACA_API_SECRET_KEY")
    if not key or not secret:
        raise AlpacaCredentialsMissing(
            "defina ALPACA_API_KEY_ID e ALPACA_API_SECRET_KEY. "
            "Use SEMPRE credenciais de paper trading; nenhuma chave real deve "
            "estar acessivel ao sistema."
        )
    return StockHistoricalDataClient(api_key=key, secret_key=secret)


def _feed() -> DataFeed:
    return DataFeed(os.environ.get("ALPACA_FEED", "iex"))


def _fmt(v) -> str:
    return str(Decimal(str(v)).quantize(_Q))


def fetch_daily_bars(
    symbols: list[str],
    start: date,
    end: date,
    asof: str | None = None,
) -> list[dict]:
    req = StockBarsRequest(
        symbol_or_symbols=symbols,
        timeframe=TimeFrame.Day,
        start=datetime.combine(start, time.min, tzinfo=timezone.utc),
        end=datetime.combine(end, time.max, tzinfo=timezone.utc),
        adjustment=Adjustment.RAW,
        feed=_feed(),
        asof=asof,
    )
    resposta = _client().get_stock_bars(req)

    linhas: list[dict] = []
    for symbol, barras in resposta.data.items():
        for b in barras:
            linhas.append(
                {
                    "symbol": symbol,
                    "session_date": b.timestamp.astimezone(timezone.utc).date().isoformat(),
                    "open": _fmt(b.open),
                    "high": _fmt(b.high),
                    "low": _fmt(b.low),
                    "close": _fmt(b.close),
                    "volume": str(int(b.volume)),
                }
            )
    return linhas


def raw_payload(rows: list[dict]) -> bytes:
    """Serializa as linhas normalizadas para o arquivo write-once."""
    return json.dumps(rows, sort_keys=True, separators=(",", ":")).encode("utf-8")
```

# arena/ingest/universe.py
"""Composicao do QQQ, point-in-time.

A composicao de um dia passado nao pode ser reconstruida depois de graca. Sem
snapshot diario, toda metrica futura sofre vies de sobrevivencia — voce mede o
desempenho de quem sobreviveu ate hoje.
"""

from __future__ import annotations

import csv
import io
from decimal import Decimal, InvalidOperation

import requests

from arena.contracts import SCHEMA_VERSION, Constituent, UniverseSnapshot

HOLDINGS_URL = (
    "https://www.invesco.com/us/financial-products/etfs/holdings/main/holdings/0"
    "?audienceType=Investor&action=download&ticker=QQQ"
)
SOURCE = "invesco_qqq_holdings_csv"
_Q = Decimal("0.0001")
_NAO_ATIVO = {"USD", "CASH", "-", ""}


def fetch_holdings_csv(timeout: int = 30) -> bytes:
    r = requests.get(HOLDINGS_URL, timeout=timeout, headers={"User-Agent": "arena/0.1"})
    r.raise_for_status()
    return r.content


def parse_holdings(payload: bytes, session_date: str) -> UniverseSnapshot:
    leitor = csv.DictReader(io.StringIO(payload.decode("utf-8-sig")))
    itens: list[Constituent] = []

    for linha in leitor:
        simbolo = (linha.get("Holding Ticker") or "").strip().upper()
        if simbolo in _NAO_ATIVO:
            continue
        try:
            peso = Decimal((linha.get("Weight") or "").strip()).quantize(_Q)
        except (InvalidOperation, ValueError):
            continue
        itens.append(Constituent(symbol=simbolo, weight_pct=str(peso)))

    if not itens:
        raise ValueError("nenhum constituinte valido no CSV de holdings")

    itens.sort(key=lambda c: (-Decimal(c.weight_pct), c.symbol))
    return UniverseSnapshot(
        schema_version=SCHEMA_VERSION,
        record_type="universe",
        session_date=session_date,
        source=SOURCE,
        constituents=itens,
    )


def top_n(snapshot: UniverseSnapshot, n: int) -> list[str]:
    return [c.symbol for c in snapshot.constituents[:n]]
```

# arena/quality/checks.py
"""Fiscal de Dados: quatro asseracoes bloqueantes.

Fronteira dura: o Fiscal DETECTA e nao decide nada. Ele emite flags tipadas;
quem aplica politica e o consumidor. Reimplementar decisao aqui e a forma mais
comum de acabar com duas regras divergentes e ninguem saber qual vale.
"""

from __future__ import annotations

from decimal import Decimal, InvalidOperation

from arena.contracts import SCHEMA_VERSION, DataQualityReport, QualityFlag

MAX_ABS_DAILY_RETURN = Decimal("0.25")
_CAMPOS = ("open", "high", "low", "close", "volume")


def run_checks(
    session_date: str,
    rows: list[dict],
    previous_closes: dict[str, Decimal],
    expected_session: str,
    required_symbols: list[str],
) -> DataQualityReport:
    flags: list[QualityFlag] = []
    quarentena: set[str] = set()

    def add(code: str, severity: str, symbol: str | None, detail: str) -> None:
        flags.append(QualityFlag(code=code, severity=severity, symbol=symbol, detail=detail))

    if not rows:
        add("NO_BARS", "block", None, f"nenhuma barra recebida para {session_date}")

    if session_date != expected_session:
        add(
            "SESSION_MISMATCH",
            "block",
            None,
            f"sessao {session_date} difere da esperada {expected_session}",
        )

    vistos: set[str] = set()
    for linha in rows:
        s = linha.get("symbol")
        vistos.add(s)

        if linha.get("session_date") != expected_session:
            add(
                "SESSION_MISMATCH",
                "block",
                s,
                f"barra datada {linha.get('session_date')}, esperada {expected_session}",
            )
            quarentena.add(s)
            continue

        nulos = [c for c in _CAMPOS if linha.get(c) in (None, "")]
        if nulos:
            add("NULL_FIELD", "block", s, f"campos nulos: {','.join(nulos)}")
            quarentena.add(s)
            continue

        try:
            valores = {c: Decimal(str(linha[c])) for c in _CAMPOS}
        except (InvalidOperation, ValueError) as exc:
            add("NULL_FIELD", "block", s, f"valor nao numerico: {exc}")
            quarentena.add(s)
            continue

        if any(valores[c] <= 0 for c in ("open", "high", "low", "close")):
            add("NON_POSITIVE_PRICE", "block", s, f"preco <= 0: {valores}")
            quarentena.add(s)
            continue

        anterior = previous_closes.get(s)
        if anterior and anterior > 0:
            variacao = (valores["close"] - anterior) / anterior
            if abs(variacao) > MAX_ABS_DAILY_RETURN:
                add(
                    "EXTREME_RETURN",
                    "warn",
                    s,
                    f"variacao diaria {variacao:.2%} acima de "
                    f"{MAX_ABS_DAILY_RETURN:.0%}; possivel acao corporativa "
                    f"nao processada",
                )
                quarentena.add(s)

    for s in required_symbols:
        if s not in vistos:
            add("MISSING_SYMBOL", "warn", s, "simbolo exigido ausente na coleta")

    return DataQualityReport(
        schema_version=SCHEMA_VERSION,
        record_type="dq",
        session_date=session_date,
        passed=not any(f.severity == "block" for f in flags),
        flags=flags,
        quarantined_symbols=sorted(quarentena),
    )
```

# arena/forecast/momentum.py
"""Grade determinística de previsões.

p_up = p_min + (p_max - p_min) * percentil_transversal(score)
score = w20 * forca_relativa_20d + w60 * forca_relativa_60d, contra o QQQ.

Coeficientes definidos A PRIORI em policy/forecast_rule.yaml e congelados antes
da primeira previsao. Nenhum deles foi ajustado a dado historico: a regra e
ingenua de proposito. Seu papel e (a) comecar a acumular amostra desde a semana
4 e (b) ser o grupo de controle permanente contra o qual os agentes de LLM do
M7 sao medidos.
"""

from __future__ import annotations

from datetime import date
from decimal import Decimal
from functools import lru_cache
from pathlib import Path

import pandas as pd
import yaml

from arena.contracts import SCHEMA_VERSION, ForecastRecord, Persona
from arena.ingest.calendar_us import session_offset

BENCHMARK = "QQQ"
RAIZ = Path(__file__).resolve().parents[2]
_Q4 = Decimal("0.0001")


@lru_cache(maxsize=1)
def load_rule() -> dict:
    return yaml.safe_load((RAIZ / "policy" / "forecast_rule.yaml").read_text(encoding="utf-8"))


def relative_strength(closes: pd.DataFrame, symbol: str, lookback: int) -> Decimal | None:
    if symbol not in closes.columns or BENCHMARK not in closes.columns:
        return None
    if len(closes) < lookback + 1:
        return None

    serie, bench = closes[symbol], closes[BENCHMARK]
    s_hoje, s_antes = serie.iloc[-1], serie.iloc[-1 - lookback]
    b_hoje, b_antes = bench.iloc[-1], bench.iloc[-1 - lookback]
    if None in (s_hoje, s_antes, b_hoje, b_antes) or s_antes == 0 or b_antes == 0:
        return None

    return (Decimal(s_hoje) / Decimal(s_antes)) / (Decimal(b_hoje) / Decimal(b_antes)) - Decimal(1)


def score_universe(closes: pd.DataFrame, symbols: list[str], rule: dict) -> dict[str, Decimal]:
    lb20 = int(rule["lookbacks"]["rel_20"])
    lb60 = int(rule["lookbacks"]["rel_60"])
    w20 = Decimal(rule["weights"]["rel_20"])
    w60 = Decimal(rule["weights"]["rel_60"])

    saida: dict[str, Decimal] = {}
    for s in symbols:
        if s == BENCHMARK:
            continue
        r20 = relative_strength(closes, s, lb20)
        r60 = relative_strength(closes, s, lb60)
        if r20 is None or r60 is None:
            continue
        saida[s] = w20 * r20 + w60 * r60
    return saida


def percentile_ranks(scores: dict[str, Decimal]) -> dict[str, Decimal]:
    if not scores:
        return {}
    if len(scores) == 1:
        return {next(iter(scores)): Decimal("0.5")}

    ordenado = sorted(scores.items(), key=lambda kv: kv[1])
    n = len(ordenado)
    posicao_media: dict[Decimal, list[int]] = {}
    for i, (_, v) in enumerate(ordenado):
        posicao_media.setdefault(v, []).append(i)

    saida: dict[str, Decimal] = {}
    for simbolo, valor in scores.items():
        idx = posicao_media[valor]
        media = Decimal(sum(idx)) / Decimal(len(idx))
        saida[simbolo] = media / Decimal(n - 1)
    return saida


def build_grid(
    session_date: date,
    closes: pd.DataFrame,
    symbols: list[str],
    universe_sha256: str,
    emitted_at_utc: str,
    rule: dict,
) -> list[ForecastRecord]:
    scores = score_universe(closes, symbols, rule)
    if not scores:
        return []

    ranks = percentile_ranks(scores)
    p_min = Decimal(rule["mapping"]["p_min"])
    p_max = Decimal(rule["mapping"]["p_max"])
    horizonte = int(rule["horizon_sessions"])
    resolucao = session_offset(session_date, horizonte).isoformat()
    fech_bench = Decimal(closes[BENCHMARK].iloc[-1])

    registros: list[ForecastRecord] = []
    for simbolo in sorted(scores):
        p_up = (p_min + (p_max - p_min) * ranks[simbolo]).quantize(_Q4)
        fechamento = Decimal(closes[simbolo].iloc[-1]).quantize(_Q4)
        for persona in Persona:
            registros.append(
                ForecastRecord(
                    schema_version=SCHEMA_VERSION,
                    record_type="forecast",
                    emitted_at_utc=emitted_at_utc,
                    session_date=session_date.isoformat(),
                    resolution_session_date=resolucao,
                    horizon_sessions=horizonte,
                    persona=persona,
                    symbol=simbolo,
                    claim="outperform_qqq",
                    p_up=str(p_up),
                    rule_version=rule["version"],
                    reference_close_raw=str(fechamento),
                    reference_close_qqq_raw=str(fech_bench.quantize(_Q4)),
                    universe_snapshot_sha256=universe_sha256,
                    outcome="pending",
                    resolved_at_utc=None,
                )
            )
    return registros
```

# arena/forecast/resolve.py
"""Resolvedor determinístico de horizonte.

Resolve SEMPRE no horizonte declarado, exista posicao ou nao — e o que impede a
amostra publicada de ser censurada exatamente nas observacoes de maior variancia.

`void` cobre os dois casos em que a comparacao de fechamentos brutos deixa de
ser valida: dado ausente, e acao corporativa nao processada dentro da janela.
Essa segunda regra e o que permite a M2 nao precisar da tabela de acoes
corporativas, que entra no M4 junto com o Ledger.
"""

from __future__ import annotations

from datetime import date
from decimal import Decimal

import pandas as pd

from arena.contracts import ForecastRecord
from arena.forecast.momentum import BENCHMARK
from arena.quality.checks import MAX_ABS_DAILY_RETURN


def _janela(closes: pd.DataFrame, inicio: date, fim: date) -> pd.DataFrame:
    return closes.loc[[i for i in closes.index if inicio <= i <= fim]].sort_index()


def _tem_salto_extremo(serie: pd.Series) -> bool:
    valores = [v for v in serie.tolist() if v is not None]
    for anterior, atual in zip(valores, valores[1:]):
        if anterior and Decimal(anterior) > 0:
            if abs((Decimal(atual) - Decimal(anterior)) / Decimal(anterior)) > MAX_ABS_DAILY_RETURN:
                return True
    return False


def resolve_one(rec: ForecastRecord, closes: pd.DataFrame, resolved_at_utc: str) -> ForecastRecord:
    inicio = date.fromisoformat(rec.session_date)
    fim = date.fromisoformat(rec.resolution_session_date)

    def _void() -> ForecastRecord:
        return rec.model_copy(update={"outcome": "void", "resolved_at_utc": resolved_at_utc})

    if rec.symbol not in closes.columns or BENCHMARK not in closes.columns:
        return _void()

    janela = _janela(closes, inicio, fim)
    if fim not in janela.index or inicio not in janela.index:
        return _void()

    s_fim, b_fim = janela.at[fim, rec.symbol], janela.at[fim, BENCHMARK]
    if s_fim is None or b_fim is None:
        return _void()

    if _tem_salto_extremo(janela[rec.symbol]) or _tem_salto_extremo(janela[BENCHMARK]):
        return _void()

    s_ini = Decimal(rec.reference_close_raw)
    b_ini = Decimal(rec.reference_close_qqq_raw)
    if s_ini <= 0 or b_ini <= 0:
        return _void()

    superou = (Decimal(s_fim) / s_ini) > (Decimal(b_fim) / b_ini)
    return rec.model_copy(
        update={"outcome": "hit" if superou else "miss", "resolved_at_utc": resolved_at_utc}
    )


def resolve_pending(
    records: list[ForecastRecord],
    closes: pd.DataFrame,
    today: date,
    resolved_at_utc: str,
) -> list[ForecastRecord]:
    saida: list[ForecastRecord] = []
    for r in records:
        if r.outcome != "pending":
            saida.append(r)
        elif date.fromisoformat(r.resolution_session_date) > today:
            saida.append(r)
        else:
            saida.append(resolve_one(r, closes, resolved_at_utc))
    return saida
```

# arena/audit/publish.py
"""Escrita dos artefatos do dia e encadeamento no hash.

Todo efeito colateral do ciclo passa por aqui. Escrita atomica via arquivo
temporario + replace: um processo interrompido no meio nunca deixa JSON
parcial encadeado.
"""

from __future__ import annotations

import os
from pathlib import Path

from arena.audit.chain import append_entry, build_entry, last_hash
from arena.canonical import canonical_bytes
from arena.contracts import DataQualityReport, ForecastRecord, OutageRecord, UniverseSnapshot


def _write_atomic(path: Path, payload: bytes) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(path.suffix + ".tmp")
    tmp.write_bytes(payload)
    os.replace(tmp, path)


def _chain_path(root: Path) -> Path:
    return root / "chain" / "CHAIN.jsonl"


def day_published(root: Path, session_date: str) -> bool:
    return (root / "data" / "forecasts" / f"{session_date}.json").exists() or (
        root / "data" / "outages" / f"{session_date}.json"
    ).exists()


def publish_day(
    root: Path,
    session_date: str,
    forecasts: list[ForecastRecord],
    dq: DataQualityReport,
    universe: UniverseSnapshot,
    created_at_utc: str,
) -> dict:
    alvos = {
        root / "data" / "forecasts" / f"{session_date}.json": [f.canonical_dict() for f in forecasts],
        root / "data" / "quality" / f"{session_date}.json": dq.canonical_dict(),
        root / "data" / "universe" / f"{session_date}.json": universe.canonical_dict(),
    }
    for caminho, conteudo in alvos.items():
        _write_atomic(caminho, canonical_bytes(conteudo))

    chain = _chain_path(root)
    entrada = build_entry(
        session_date, created_at_utc, sorted(alvos.keys()), last_hash(chain), root
    )
    append_entry(chain, entrada)
    return entrada


def publish_outage(root: Path, outage: OutageRecord, created_at_utc: str) -> dict:
    caminho = root / "data" / "outages" / f"{outage.session_date}.json"
    _write_atomic(caminho, canonical_bytes(outage.canonical_dict()))

    chain = _chain_path(root)
    entrada = build_entry(
        outage.session_date, created_at_utc, [caminho], last_hash(chain), root
    )
    append_entry(chain, entrada)
    return entrada

# arena/cycle.py
"""Ciclo diário.

Estrutura deliberada: `daily_cycle` recebe suas dependencias de IO por
parametro (`deps`), o que a torna testavel offline por completo e prepara o
Champion/Challenger do M16 — que precisa rodar duas politicas sobre o MESMO
snapshot de dados.

Regra dura: falha em qualquer estagio publica OutageRecord e encadeia. Um dia
sem registro e indistinguivel de uma decisao omitida.
"""

from __future__ import annotations

import traceback
from datetime import date
from decimal import Decimal
from pathlib import Path

from arena.audit.publish import day_published, publish_day, publish_outage
from arena.contracts import SCHEMA_VERSION, OutageRecord
from arena.forecast.momentum import BENCHMARK, build_grid, load_rule
from arena.ingest.calendar_us import is_session, last_n_sessions
from arena.ingest.universe import parse_holdings, top_n
from arena.quality.checks import run_checks
from arena.storage.archive import archive_raw
from arena.storage.bars import read_bars_as_of, write_bars

LOOKBACK_SESSOES = 70


def _outage(root: Path, session_date: date, stage: str, reason: str, detail: str, now: str) -> dict:
    rec = OutageRecord(
        schema_version=SCHEMA_VERSION,
        record_type="outage",
        detected_at_utc=now,
        session_date=session_date.isoformat(),
        stage=stage,
        reason=reason,
        detail=detail[:2000],
    )
    entrada = publish_outage(root, rec, now)
    return {"status": "outage", "entry": entrada, "n_forecasts": 0}


def daily_cycle(root: Path, session_date: date, deps: dict, now_utc: str) -> dict:
    d = session_date.isoformat()

    if day_published(root, d):
        return {"status": "already_published", "entry": None, "n_forecasts": 0}

    if not is_session(session_date):
        return _outage(root, session_date, "calendar", "NOT_A_SESSION", f"{d} nao e pregao", now_utc)

    # 1. universo point-in-time
    try:
        payload = deps["fetch_holdings"]()
        archive_raw(root, "invesco_qqq", d, now_utc, payload, "text/csv")
        universo = parse_holdings(payload, d)
    except Exception as exc:
        return _outage(
            root, session_date, "universe", "UNIVERSE_UNAVAILABLE",
            f"{exc}\n{traceback.format_exc()}", now_utc,
        )

    rule = load_rule()
    simbolos = top_n(universo, int(rule["universe_top_n"]))
    pedidos = sorted(set(simbolos) | {BENCHMARK})

    # 2. barras
    sessoes = last_n_sessions(session_date, LOOKBACK_SESSOES)
    try:
        linhas = deps["fetch_bars"](pedidos, sessoes[0], session_date)
        archive_raw(
            root, "alpaca_bars", d, now_utc,
            __import__("json").dumps(linhas, sort_keys=True, separators=(",", ":")).encode(),
            "application/json",
        )
    except Exception as exc:
        return _outage(
            root, session_date, "ingest", "DATA_UNAVAILABLE",
            f"{exc}\n{traceback.format_exc()}", now_utc,
        )

    for sessao in {r["session_date"] for r in linhas}:
        write_bars(root, [r for r in linhas if r["session_date"] == sessao])

    # 3. Fiscal
    do_dia = [r for r in linhas if r["session_date"] == d]
    anterior = sessoes[-2] if len(sessoes) >= 2 else None
    fechamentos_anteriores: dict[str, Decimal] = {}
    if anterior:
        for r in linhas:
            if r["session_date"] == anterior.isoformat():
                fechamentos_anteriores[r["symbol"]] = Decimal(r["close"])

    dq = run_checks(d, do_dia, fechamentos_anteriores, d, pedidos)
    if not dq.passed:
        codigos = ",".join(sorted({f.code for f in dq.flags if f.severity == "block"}))
        return _outage(root, session_date, "quality", "QUALITY_BLOCK", codigos, now_utc)

    # 4. grade
    elegiveis = [s for s in simbolos if s not in dq.quarantined_symbols]
    closes = read_bars_as_of(root, sorted(set(elegiveis) | {BENCHMARK}), sessoes)
    grade = build_grid(session_date, closes, elegiveis, universo.sha256(), now_utc, rule)

    if not grade:
        return _outage(
            root, session_date, "forecast", "EMPTY_GRID",
            "historico insuficiente para 60 sessoes de lookback", now_utc,
        )

    entrada = publish_day(root, d, grade, dq, universo, now_utc)
    return {"status": "published", "entry": entrada, "n_forecasts": len(grade)}

    ap.add_argument("comando", choices=["run-daily", "verify"])
    ap.add_argument("--date")

if __name__ == "__main__":

# scripts/build_site.py
"""Monta o diretorio publicavel do GitHub Pages.

O painel le EXCLUSIVAMENTE os JSON commitados. Nao existe banco, nao existe
servidor de aplicacao, e nao existe caminho de leitura mutavel entre o registro
e o publico — o que e feature de credibilidade, nao limitacao.
"""

from __future__ import annotations

import json
import shutil
from datetime import datetime, timezone
from pathlib import Path


def build_index(root: Path) -> dict:
    sessoes: list[dict] = []
    total_previsoes = 0

    for arquivo in sorted((root / "data" / "forecasts").glob("*.json")):
        registros = json.loads(arquivo.read_text(encoding="utf-8"))
        total_previsoes += len(registros)
        sessoes.append({"date": arquivo.stem, "type": "forecast", "n": len(registros)})

    outages = root / "data" / "outages"
    if outages.exists():
        for arquivo in sorted(outages.glob("*.json")):
            corpo = json.loads(arquivo.read_text(encoding="utf-8"))
            sessoes.append(
                {"date": arquivo.stem, "type": "outage", "reason": corpo.get("reason", "?")}
            )

    sessoes.sort(key=lambda s: s["date"])
    return {
        "generated_at_utc": datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"),
        "sessions": sessoes,
        "counts": {
            "days_published": len(sessoes),
            "days_outage": sum(1 for s in sessoes if s["type"] == "outage"),
            "forecasts_total": total_previsoes,
            "first_session": sessoes[0]["date"] if sessoes else None,
            "last_session": sessoes[-1]["date"] if sessoes else None,
        },
    }


def assemble(root: Path, out: Path) -> None:
    if out.exists():
        shutil.rmtree(out)
    out.mkdir(parents=True)

    shutil.copytree(root / "site", out, dirs_exist_ok=True)
    shutil.copytree(root / "data", out / "data", dirs_exist_ok=True)
    shutil.copytree(root / "chain", out / "chain", dirs_exist_ok=True)

    for nome in ("MANIFESTO.md", "README.md"):
        origem = root / nome
        if origem.exists():
            shutil.copy2(origem, out / nome)

    (out / "data" / "index.json").write_text(
        json.dumps(build_index(root), indent=2, ensure_ascii=False), encoding="utf-8"
    )


if __name__ == "__main__":
    raiz = Path(__file__).resolve().parents[1]
    assemble(raiz, raiz / "_site")
    print(f"_site pronto: {json.dumps(build_index(raiz)['counts'], ensure_ascii=False)}")
```

// site/verify.js
// Recomputa a cadeia de hash no navegador do visitante.
// A forma canonica precisa ser BYTE-IDENTICA a de arena/canonical.py:
// chaves ordenadas, sem espacos, UTF-8 literal, e nenhum numero decimal
// (todo valor decimal viaja como string, justamente para que os dois
// runtimes concordem sem ambiguidade de formatacao de ponto flutuante).

export function canonicalize(v) {
  if (Array.isArray(v)) return "[" + v.map(canonicalize).join(",") + "]";
  if (v !== null && typeof v === "object") {
    return (
      "{" +
      Object.keys(v)
        .sort()
        .map((k) => JSON.stringify(k) + ":" + canonicalize(v[k]))
        .join(",") +
      "}"
    );
  }
  if (typeof v === "number" && !Number.isInteger(v)) {
    throw new Error("numero decimal em registro hasheado: " + v);
  }
  return JSON.stringify(v);
}

export async function sha256Hex(text) {
  const buf = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(text));
  return [...new Uint8Array(buf)].map((b) => b.toString(16).padStart(2, "0")).join("");
}

const GENESIS = "0".repeat(64);

export async function verifyChain(basePath = ".") {
  const texto = await (await fetch(`${basePath}/chain/CHAIN.jsonl`)).text();
  const entradas = texto.split("\n").filter((l) => l.trim()).map((l) => JSON.parse(l));

  const erros = [];
  let esperadoPrev = GENESIS;

  for (const [i, entrada] of entradas.entries()) {
    const rotulo = `linha ${i + 1} (${entrada.session_date})`;

    const corpo = { ...entrada };
    delete corpo.entry_hash;
    if ((await sha256Hex(canonicalize(corpo))) !== entrada.entry_hash) {
      erros.push(`${rotulo}: entry_hash divergente`);
    }
    if (entrada.prev_hash !== esperadoPrev) {
      erros.push(`${rotulo}: prev_hash nao aponta para a entrada anterior`);
    }

    for (const arq of entrada.files) {
      try {
        const bytes = await (await fetch(`${basePath}/${arq.path}`)).arrayBuffer();
        const h = [...new Uint8Array(await crypto.subtle.digest("SHA-256", bytes))]
          .map((b) => b.toString(16).padStart(2, "0"))
          .join("");
        if (h !== arq.sha256) erros.push(`${rotulo}: sha256 divergente em ${arq.path}`);
      } catch {
        erros.push(`${rotulo}: arquivo inacessivel ${arq.path}`);
      }
    }

    esperadoPrev = entrada.entry_hash;
  }

  return { entries: entradas.length, errors: erros };
}
```

Duas implementacoes da mesma serializacao divergindo em silencio faria o
verificador publico acusar fraude onde nao houve — o que destruiria a

import shutil
import subprocess
import sys
import tempfile
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))

from arena.canonical import sha256_hex  # noqa: E402

    {"a": "1", "b": "2"},

    {"t": "ação e çedilha — travessão"},
    {"n": 5, "ok": True, "nada": None},
    {"files": [{"path": "data/x.json", "sha256": "a" * 64}], "prev_hash": "0" * 64},
    {"multi": "linha1\nlinha2\ttab \"citada\" e \\barra"},
]

const casos = JSON.parse(process.argv[2]);
const out = [];

console.log(JSON.stringify(out));

    ]
